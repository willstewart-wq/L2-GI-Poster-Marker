# ==========================================
# 2. FILE EXTRACTION UTILITIES
# ==========================================
def extract_document_content(uploaded_file):
    """Extracts text, page count, and image data from uploaded files."""
    file_type = uploaded_file.name.split(".")[-1].lower()
    text = ""
    page_count = 1
    image_bytes_list = []

    if file_type == "pdf":
        reader = PdfReader(uploaded_file)
        page_count = len(reader.pages)
        for idx, page in enumerate(reader.pages):
            text += f"\n--- Page {idx + 1} ---\n" + (page.extract_text() or "")
            for img in page.images:
                image_bytes_list.append(img.data)
    elif file_type in ["png", "jpg", "jpeg"]:
        page_count = 1
        img_data = uploaded_file.read()
        image_bytes_list.append(img_data)
        text = "[Image submission loaded for vision evaluation]"

    return {
        "text": text,
        "page_count": page_count,
        "image_count": len(image_bytes_list),
        "images": image_bytes_list
    }

# 3. AI EVALUATION ENGINE
# ==========================================
def mark_submission(api_key, poster_title, rubric, parsed_doc):
    client = openai.OpenAI(api_key=api_key)

    prompt = f"""
    You are an expert assessor marking student poster submissions for fitness qualifications.
    
    ASSIGNMENT: "{poster_title}"
    
    PAGE CONSTRAINT: Maximum 2 sides of A4 allowed.
    DETECTED PAGE COUNT: {parsed_doc['page_count']} page(s).
    IMAGES DETECTED: {parsed_doc['image_count']} image(s).
    
    REQUIRED CRITERIA TO MARK AGAINST:
    {json.dumps(rubric['criteria'], indent=2)}
    
    EXTRACTED TEXT FROM WORK:
    {parsed_doc['text']}
    
    INSTRUCTIONS:
    1. Check if the work is within the 2-page A4 limit.
    2. Check if relevant images and bullet points/concise formatting are used.
    3. Evaluate each numbered criterion as "Met", "Partially Met", or "Not Met".
    4. Provide clear evidence from the text for what HAS been met.
    5. Provide specific, actionable guidance on WHAT STILL NEEDS TO BE DONE.
    
    Return strict JSON matching this structure ONLY:
    {{
        "page_limit_met": true/false,
        "has_images": true/false,
        "has_references": true/false,
        "formatting_notes": "Brief comment on page count, concise bulleted style, and images.",
        "evaluations": [
            {{
                "id": 1,
                "title": "Criterion Title",
                "status": "Met" | "Partially Met" | "Not Met",
                "evidence_met": "Quote or summary of what student completed.",
                "action_needed": "What still needs to be done to meet the criteria."
            }}
        ],
        "overall_summary": "Encouraging summary highlighting strengths and missing requirements."
    }}
    """

    messages = [{"role": "system", "content": "You are a professional assessor. Output strictly valid JSON."},
                {"role": "user", "content": prompt}]

    # Include first image if provided for visual verification
    if parsed_doc["images"]:
        b64_img = base64.b64encode(parsed_doc["images"][0]).decode("utf-8")
        messages.append({
            "role": "user",
            "content": [
                {"type": "text", "text": "Student poster page preview:"},
                {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{b64_img}"}}
            ]
        })

    response = client.chat.completions.create(
        model="gpt-4o",
        messages=messages,
        response_format={"type": "json_object"},
        temperature=0.1
    )
    return json.loads(response.choices[0].message.content)

                    else:
                        st.error(f"📄 Page Limit: {parsed_doc['page_count']} / 2 Sides (Exceeded)")
                with m2:
                    if result.get("has_images") or parsed_doc["image_count"] > 0:
                        st.success(f"🖼️ Images Included ({parsed_doc['image_count']} found)")
                    else:
                        st.warning("🖼️ No Images Detected")
                with m3:
                    if result.get("has_references"):
                        st.success("📚 References Found")
                    else:
                        st.error("📚 References Missing")

                st.info(f"**Formatting & Layout Feedback:** {result.get('formatting_notes', '')}")

                # Criteria Breakdown Table
                st.subheader("🎯 Criteria Completion Checklist")
                
                evals = result.get("evaluations", [])
                met_cnt = sum(1 for e in evals if e["status"] == "Met")
                st.progress(met_cnt / len(evals) if evals else 0, text=f"Met {met_cnt} of {len(evals)} required criteria")

                for item in evals:
                    status = item["status"]
                    icon = "✅" if status == "Met" else ("⚠️" if status == "Partially Met" else "❌")
                    
                    with st.expander(f"{icon} **Criterion {item['id']}: {item['title']}** — *{status}*"):
                        c1, c2 = st.columns(2)
                        with c1:
                            st.markdown("**What Has Been Met:**")
                            st.write(item.get("evidence_met", "None found."))
                        with c2:
                            st.markdown("**What Still Needs To Be Done:**")
                            st.write(item.get("action_needed", "Complete."))

                # Overall Action Plan
                st.subheader("📝 Summary & Next Steps for Student")
                st.success(result.get("overall_summary", "Evaluation complete."))
